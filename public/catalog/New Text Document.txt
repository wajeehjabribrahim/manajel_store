    let currentLang = 'ar';
    function updateLanguageContent() {
        const htmlTag = document.getElementById('mainHtml');
        htmlTag.dir = currentLang === 'ar' ? 'rtl' : 'ltr';
        document.querySelectorAll('[data-key]').forEach(el => {
            const key = el.getAttribute('data-key');
            const value = langData[currentLang][key];
            if (el.classList.contains('description')) {
                el.innerText = value || '';
            } else {
                el.textContent = value || '';
            }
        });
        document.getElementById('langBtn').textContent = langData[currentLang].langBtn;
        if (currentLang === 'ar') {
            document.body.classList.remove('english');
            document.body.style.fontFamily = "'Cairo', sans-serif";
        } else {
            document.body.classList.add('english');
            document.body.style.fontFamily = "'Tajawal-Bold', 'Inter', sans-serif";
        }
        document.querySelectorAll('.cover-page').forEach(el => {
            el.style.fontFamily = "'Cairo', sans-serif";
        });
    }

    function toggleLanguage() {
        currentLang = currentLang === 'ar' ? 'en' : 'ar';
        updateLanguageContent();
    }

    // Ensure content is set on initial load
    document.addEventListener('DOMContentLoaded', updateLanguageContent);
